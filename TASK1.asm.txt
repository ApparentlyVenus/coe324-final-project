; export symbols
            XDEF Entry, _Startup            ; export 'Entry' symbol
            ABSENTRY Entry        ; for absolute assembly: mark this as application entry point



; Include derivative-specific definitions 
		INCLUDE 'derivative.inc' 

ROMStart    EQU  $4000  ; absolute address to place my code/constant data

; variable/data section

            ORG RAMStart
 ; Insert here your data definition.
 LED_MASK DS.B 1  ;Tracks the active LED
 TASK_STATE DS.B 1  ;1=Task 1, 2=Task 2


; code section
            ORG   ROMStart


Entry:
_Startup:
            LDS   #RAMEnd+1       ; initialize the stack pointer

            CLI                     ; enable interrupts
INIT_TASK1:
            MOVB #$F0, DDRB
            MOVB #$C0, ATD0CTL2
            JSR  DELAY20
            MOVB #$08, ATD0CTL3
            MOVB #$80, ATD0CTL4
            MOVB #$A5, ATD0CTL5
            MOVB #$80, LED_MASK     ;Start at MSB for right shift
            RTS
            
EXECUTE_TASK1:
WAIT_ATD:
            BRCLR ATD0STAT0, #$80, WAIT_ATD
            LDAA ATD0DR0L      ;Load 8 bit potentiometer value int ACC A
            LDAB LED_MASK     ;Load LED state
            LSRB
            CMPB #$08              ;Shift right
            BNE UPDATE_LED    ;If it hasn't fallen below bit 0 -> skip wrap around
            LDAB #$80         ;Wrap around back to the leftmost LED
            
UPDATE_LED:
            STAB LED_MASK     ;Save the shifted state
            COMB              ;Invert logic so active low 
            STAB PORTB
            JSR POT_DELAY     ;Acc A contains the ATD multiplier
            
            LDAB TASK_STATE
            CMPB #1
            BEQ EXECUTE_TASK1  ;If still Task 1, keep looping
            RTS
            
POT_DELAY:
            CMPA #50
            BHS START_PD
            LDAA #50
START_PD:
            PSHA                  ;Save the potentiometer value
PD:
            LDX #105
AGAIN1:
            PSHB
            PULB
            PSHB
            PULB
            PSHB
            PULB
            DEX
            BNE AGAIN1
            
            DECA
            BNE PD 
            PULA                  ;Restore the potentiometer value
            RTS
            
DELAY20:
            LDX #4
AGAIN2:     
            PSHB
            PULB
            DEX
            BNE AGAIN2
            RTS           
      
            
;**************************************************************
;*                 Interrupt Vectors                          *
;**************************************************************
            ORG   $FFFE
            DC.W  Entry           ; Reset Vector
